To-Do (MVC)

    Realizar un resumen del Modelo MVC - Incluir Ejemplo

    MVC es un patrón de arquitectura que separa una aplicación en tres componentes principales: Model (datos y lógica), View (presentación) y Controller (orquesta la interacción). Esta separación mejora la organización, facilita pruebas y permite que equipos trabajen en distintas partes sin pisarse.

    Responsabilidades

    • Model: representa los datos y la lógica de negocio. Notifica cambios a quien esté suscrito.
    • View: muestra los datos al usuario y captura eventos de UI (clics, formularios). No debe contener lógica de negocio.
    • Controller: recibe eventos de la View, actualiza el Model y normalmente pide a la View que se refresque.

    Flujo típico: View → Controller → Model → (Model notifica) → View se actualiza.

    Ventajas / Cuándo usarlo (Resumido)

    • Mejora mantenibilidad y testeo.
    • Separación clara de responsabilidades.
    • Muy útil en apps con UI rica o equipos grandes.
    • No siempre necesario para apps muy pequeñas (overhead).

    Ejemplo práctico: mini-app To-Do (HTML + JavaScript)

    Ejemplo práctico: mini-app To-Do (HTML + JavaScript). Código de ejemplo:

    <!doctype html>
<html>
<head>
  <meta charset="utf-8" />
  <title>MVC - Mini ToDo</title>
  <style>
    body { font-family: Arial; max-width: 540px; margin: 40px auto; }
    ul { padding-left: 20px; }
    li { margin: 6px 0; }
    button { margin-left: 8px; }
  </style>
</head>
<body>
  <h1>To-Do (MVC)</h1>

  <input id="newTodo" placeholder="Nueva tarea..." />
  <button id="addBtn">Añadir</button>

  <ul id="list"></ul>

  <script>
    // MODEL
    class TodoModel {
      constructor() {
        this.todos = [];            // datos
        this.onChange = () => {};   // callback de notificación
      }
      bindOnChange(callback) { this.onChange = callback; }
      add(text) {
        if (!text) return;
        const item = { id: Date.now(), text, done: false };
        this.todos.push(item);
        this.onChange(this.todos);
      }
      remove(id) {
        this.todos = this.todos.filter(t => t.id !== id);
        this.onChange(this.todos);
      }
      toggle(id) {
        this.todos = this.todos.map(t => t.id === id ? {...t, done: !t.done} : t);
        this.onChange(this.todos);
      }
    }

    // VIEW
    class TodoView {
      constructor() {
        this.input = document.getElementById('newTodo');
        this.addBtn = document.getElementById('addBtn');
        this.list = document.getElementById('list');
      }

      render(todos) {
        this.list.innerHTML = '';
        todos.forEach(t => {
          const li = document.createElement('li');
          li.dataset.id = t.id;
          li.innerHTML = `
            <label style="text-decoration:${t.done ? 'line-through' : 'none'}">
              <input type="checkbox" ${t.done ? 'checked' : ''}/> ${this._escape(t.text)}
            </label>
            <button class="del">Eliminar</button>
          `;
          this.list.appendChild(li);
        });
      }

      _escape(s) {
        const div = document.createElement('div');
        div.textContent = s;
        return div.innerHTML;
      }

      bindAdd(handler) {
        this.addBtn.addEventListener('click', () => {
          const value = this.input.value.trim();
          if (value) {
            handler(value);
            this.input.value = '';
            this.input.focus();
          }
        });
        this.input.addEventListener('keydown', (e) => {
          if (e.key === 'Enter') this.addBtn.click();
        });
      }

      bindToggleAndRemove(handlerToggle, handlerRemove) {
        this.list.addEventListener('click', (e) => {
          const li = e.target.closest('li');
          if (!li) return;
          const id = Number(li.dataset.id);
          if (e.target.classList.contains('del')) {
            handlerRemove(id);
          } else if (e.target.type === 'checkbox') {
            handlerToggle(id);
          }
        });
      }
    }

    // CONTROLLER
    class TodoController {
      constructor(model, view) {
        this.model = model;
        this.view = view;

        // cuando el model cambia, renderiza la vista
        this.model.bindOnChange((todos) => this.view.render(todos));

        // enlaza eventos de la vista con acciones en el model
        this.view.bindAdd(text => this.model.add(text));
        this.view.bindToggleAndRemove(
          id => this.model.toggle(id),
          id => this.model.remove(id)
        );

        // render inicial
        this.view.render(this.model.todos);
      }
    }

    // Inicialización
    const app = new TodoController(new TodoModel(), new TodoView());
  </script>
</body>
</html>

    Qué muestra el ejemplo

    • Model guarda la lista de tareas y notifica cambios mediante onChange.
    • View renderiza la UI y expone métodos para enlazar eventos (añadir, eliminar, marcar).
    • Controller conecta View ↔ Model: cuando el usuario añade una tarea, el controller llama model.add(...) y el model notifica para que la view se refresque.
